-- Top customers by revenue, last 90 days
SELECT u.id,
u.name,
u.email,
COUNT(o.id) AS orders,
ROUND(SUM(o.total), 2) AS lifetime_value
FROM users u
JOIN orders o ON o.user_id = u.id
WHERE o.placed_at >= DATE('now', '-90 days') AND o.status = 'paid'
GROUP BY u.id ORDER BY lifetime_value DESC LIMIT 50;
